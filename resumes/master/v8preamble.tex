% ---- KNOBS: length registers, retunable from the body. Ranges are hard. Move one at a time.
\newlength{\SectionGapTop}\setlength{\SectionGapTop}{5pt}   % 3pt - 7pt
\newlength{\SectionGapBot}\setlength{\SectionGapBot}{2pt}   % 1pt - 3pt
\newlength{\ItemGap}\setlength{\ItemGap}{1.8pt}             % 0.8pt - 2.2pt. 1.8 is the
% verified default: same content and page count as 1.2pt, but page 1 drops 100->92% fill and
% page 2 rises 70->80%, so the document reads less crammed and the pages balance.
\newlength{\BlockGap}\setlength{\BlockGap}{3pt}             % 2pt - 5pt

\titleformat{\section}{\scshape\large}{}{0pt}{}[\vspace{-4pt}\rule{\textwidth}{0.4pt}]
